No new gate is added — the menu is screenshot-verified, the decoders host-tested.

**Architecture:** A visual reskin over the working Phase-3 scene logic, unblocked by three self-contained decoder additions. Task 1 is a confirmatory RE-GATE calibration (the hunt `docs/re/mainmenu.md` is COMPLETE — every menu art id is CONFIRMED present; T1 re-probes via `ra2extract` + a quick PNG render to close the four `[RE-FALLBACK]` items — the correct menu palette that resolves the dark render, the approximate button/logo coords, the exact menu-item list + live/greyed set, the `grinder.wav` track + loop — NOT a fresh hunt). // A decoded 8-bit palettized PCX image. `indices` is w*h palette indices
```cpp
// (row-major, top-down); `palette` is the FULL 8-bit trailer palette (a=255).
// RA2's logo.pcx/title.pcx/loadscreens are all 8-bit RLE PCX with the palette
```

```cpp
#include "ra2assets/pcx.hpp"
#include <cstring>
#include <stdexcept>

namespace ra2assets {
namespace {
uint16_t rd16(const uint8_t* p) { return static_cast<uint16_t>(p[0] | (p[1] << 8)); }
}  // namespace

PcxImage decode_pcx(const uint8_t* d, std::size_t len) {
    if (len < 128) throw std::runtime_error("pcx: shorter than header");
    if (d[0] != 0x0A) throw std::runtime_error("pcx: bad manufacturer (!=0x0A)");
    const uint8_t encoding = d[2];      // 1 = RLE
    const uint8_t bpp = d[3];           // bits per pixel per plane
    if (encoding != 1) throw std::runtime_error("pcx: not RLE-encoded");
    const int xmin = rd16(d + 4), ymin = rd16(d + 6);
    const int xmax = rd16(d + 8), ymax = rd16(d + 10);
    const uint8_t planes = d[65];
    const int bpl = rd16(d + 66);       // bytes per scanline per plane (may pad)
    if (bpp != 8 || planes != 1)
        throw std::runtime_error("pcx: only 8-bit single-plane supported");
    PcxImage img;
    img.w = xmax - xmin + 1;
    img.h = ymax - ymin + 1;
    if (img.w <= 0 || img.h <= 0) throw std::runtime_error("pcx: bad dimensions");
    img.indices.assign(static_cast<size_t>(img.w) * img.h, 0);

    // RLE body starts at 128. Each scanline decodes bpl bytes; the first `w`
    // are the pixels (the rest is padding, discarded).
    std::size_t pos = 128;
    for (int row = 0; row < img.h; ++row) {
        int x = 0;
        while (x < bpl) {
            if (pos >= len) throw std::runtime_error("pcx: truncated RLE body");
            uint8_t b = d[pos++];
            int count = 1;
            if ((b & 0xC0) == 0xC0) {           // run: low 6 bits = count
                count = b & 0x3F;
                if (pos >= len) throw std::runtime_error("pcx: truncated RLE run");
                b = d[pos++];
            }
            for (int k = 0; k < count && x < bpl; ++k, ++x)
                if (x < img.w) img.indices[static_cast<size_t>(row) * img.w + x] = b;
        }
    }

    // Trailer palette: last 769 bytes = 0x0C marker + 256*3 RGB (FULL 8-bit,
    // NOT 6-bit VGA -- do NOT route through Palette::load, which does v<<2).
    if (len < 769) throw std::runtime_error("pcx: no trailer palette");
    const uint8_t* pal = d + (len - 769);
    if (pal[0] != 0x0C) throw std::runtime_error("pcx: trailer palette marker != 0x0C");
    const uint8_t* rgb = pal + 1;
    for (int i = 0; i < 256; ++i)
        img.palette[i] = Rgba{rgb[i * 3 + 0], rgb[i * 3 + 1], rgb[i * 3 + 2], 255};
    return img;
}

std::vector<uint8_t> PcxImage::rgba() const {
    std::vector<uint8_t> out(static_cast<size_t>(w) * h * 4);
    for (size_t i = 0; i < indices.size(); ++i) {
        const Rgba& c = palette[indices[i]];
        out[i * 4 + 0] = c.r; out[i * 4 + 1] = c.g; out[i * 4 + 2] = c.b; out[i * 4 + 3] = 255;
    }
    return out;
}

}  // namespace ra2assets
```

```cpp
namespace ra2assets {

// One decoded glyph: w*h palette indices (row-major, index 0 = transparent).
struct FntGlyph {

// A Westwood .fnt variable-width bitmap font (RA2 shell font, e.g.
// metal12.fnt / 12metfnt.fnt). Glyphs are indexed by byte value in
// [first(), last()]; width() is the per-glyph advance. [RE: mainmenu.md §3]
class Font {
public:
    static Font load(std::vector<uint8_t> bytes);
    int first() const { return first_; }
    int last()  const { return last_; }
    int count() const { return last_ - first_ + 1; }
    int height() const { return height_; }        // max glyph height
    int max_width() const { return max_width_; }
    int width(int ch) const;                       // advance width; 0 if out of range
    const FntGlyph& glyph(int ch) const;           // throws if out of range

private:

    std::vector<FntGlyph> glyphs_;                 // index ch-first_
    std::vector<int> widths_;                      // index ch-first_
```

```cpp
#include "ra2assets/fnt.hpp"
#include <stdexcept>

namespace ra2assets {
namespace {
uint16_t rd16(const uint8_t* p) { return static_cast<uint16_t>(p[0] | (p[1] << 8)); }
}  // namespace

Font Font::load(std::vector<uint8_t> b) {
    if (b.size() < 0x14) throw std::runtime_error("fnt: shorter than header");
    const uint8_t* d = b.data();
    // Westwood font header (XCC-documented; confirmed by the T3 Step 1 probe):
    //   0x00 u16 fileSize   0x05 u16 dataBlockOffsets  0x08 u16 widthOffset
    //   0x0A u16 heightOffset  0x0E u8 first  0x0F u8 last(count)  0x10 u8 height
    //   0x11 u8 maxWidth        (offsets are absolute byte offsets into the file)
    const uint16_t off_data   = rd16(d + 0x05);   // table of u16 per-glyph data offsets
    const uint16_t off_width  = rd16(d + 0x08);   // table of u8  per-glyph widths
    const uint16_t off_height = rd16(d + 0x0A);   // table of u8 pairs {y-offset, rows}
    Font f;
    f.first_     = d[0x0E];
    const int n  = d[0x0F];                        // glyph count (last index-ish)
    f.height_    = d[0x10];
    f.max_width_ = d[0x11];
    f.last_      = f.first_ + n - 1;
    f.glyphs_.resize(n);
    f.widths_.resize(n);
    for (int i = 0; i < n; ++i) {
        const int w = d[off_width + i];
        const uint8_t* hh = d + off_height + i * 2;
        const int yoff = hh[0], rows = hh[1];
        const uint16_t gdo = rd16(d + off_data + i * 2);   // this glyph's bitmap offset
        f.widths_[i] = w;
        FntGlyph& g = f.glyphs_[i];
        g.w = w;
        g.h = f.height_;
        g.pixels.assign(static_cast<size_t>(g.w) * g.h, 0);
        // Glyph rows are packed 4-bit-per-pixel (2 px/byte), `rows` rows tall,
        // starting at vertical y-offset `yoff`. Width in bytes = (w+1)/2.
        const int wbytes = (w + 1) / 2;
        const uint8_t* src = d + gdo;
        for (int ry = 0; ry < rows; ++ry) {
            for (int rx = 0; rx < w; ++rx) {
                const uint8_t byte = src[ry * wbytes + rx / 2];
                const uint8_t idx = (rx & 1) ? (byte >> 4) : (byte & 0x0F);
                g.pixels[static_cast<size_t>(yoff + ry) * g.w + rx] = idx;
            }
        }
    }
    return f;
}

int Font::width(int ch) const {
    if (ch < first_ || ch > last_) return 0;
    return widths_[ch - first_];
}

const FntGlyph& Font::glyph(int ch) const {
    if (ch < first_ || ch > last_) throw std::runtime_error("fnt: glyph out of range");
    return glyphs_[ch - first_];
}

}  // namespace ra2assets
```

```cpp
// the TextureCache's ready-RGBA path (index-0-transparent does NOT apply --
// PCX logos use index 0 as a real color). Cached by `key` (position-indep).
void blit_pcx(SDL_Renderer* ren, ra2engine::TextureCache& tc, const std::string& key,
              const ra2assets::PcxImage& img, int x, int y) {
    if (img.w <= 0 || img.h <= 0) return;

    if (!tex) {
```
