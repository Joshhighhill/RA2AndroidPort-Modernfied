```cpp
    std::string id;
    std::string name;   // Name=
```

```cpp
#include "aimd_parse.hpp"
#include <cctype>
#include <cstdlib>
#include <unordered_map>

namespace ra2engine {
namespace {

std::string upper(std::string s) {
    for (char& c : s) c = (char)std::toupper((unsigned char)c);
    return s;
}
// RA2 boolean values: yes/no (also true/1 tolerated) [RE: ai-team.md §1].
bool truthy(const std::string& v) {
    if (v.empty()) return false;
    const char c = (char)std::toupper((unsigned char)v[0]);
    return c == 'Y' || c == 'T' || c == '1';
}
bool all_digits(const std::string& s) {
    if (s.empty()) return false;
    for (char c : s)
        if (!std::isdigit((unsigned char)c)) return false;
    return true;
}
// "count,UNITID" / "action,argument": split on the FIRST comma.
bool split2(const std::string& v, std::string& a, std::string& b) {
    const std::size_t comma = v.find(',');
    if (comma == std::string::npos) return false;
    a = v.substr(0, comma);
    b = v.substr(comma + 1);
    return true;
}

}  // namespace

int waypoint_from_letter(const std::string& s) {
    // Bijective base-26: A=0..Z=25, AA=26 [RE-GATE APPLIED: ai-team.md §1].
    if (s.empty()) return -1;
    long v = 0;
    for (char c : s) {
        const char u = (char)std::toupper((unsigned char)c);
        if (u < 'A' || u > 'Z') return -1;
        v = v * 26 + (u - 'A' + 1);
    }
    return (int)(v - 1);
}

AimdData parse_aimd(const ra2assets::Ini& ini,
                    const std::function<int(const std::string&)>& unit_index) {
    AimdData d;
    std::unordered_map<std::string, int> tf_ix, sc_ix;  // entity id -> table index

    // [TaskForces] index section: N=id, file order [RE: ai-team.md §1].
    for (const auto& kv : ini.section("TaskForces")) {
        const std::string id = upper(kv.second);
        if (id.empty() || !ini.has(id) || tf_ix.count(id)) continue;  // dangling/dup
        TaskForce tf;
        tf.id = id;
        bool ok = true;
        for (const auto& e : ini.section(id)) {
            if (e.first == "NAME") tf.name = e.second;
            else if (e.first == "GROUP") tf.group = std::atoi(e.second.c_str());
            else if (all_digits(e.first)) {   // N=count,UNITID
                std::string cnt, uid;
                if (!split2(e.second, cnt, uid)) { ok = false; break; }
                TaskForceMember m;
                m.count = std::atoi(cnt.c_str());
                m.unit = unit_index(upper(uid));
                if (m.count <= 0 || m.unit < 0) { ok = false; break; }  // art-guard-drop
                tf.members.push_back(m);
            }
        }
        if (!ok || tf.members.empty()) continue;  // whole TaskForce dropped
        tf_ix[id] = (int)d.taskforces.size();
        d.taskforces.push_back(std::move(tf));
    }

    // [ScriptTypes] index section: N=id [RE: ai-team.md §1/§3]. Every action is
    // KEPT even if this engine interprets it inert — the inert split is the
    // INTERPRETER's job (ai_team), not the parser's.
    for (const auto& kv : ini.section("ScriptTypes")) {
        const std::string id = upper(kv.second);
        if (id.empty() || !ini.has(id) || sc_ix.count(id)) continue;
        ScriptType sc;
        sc.id = id;
        for (const auto& e : ini.section(id)) {
            if (e.first == "NAME") sc.name = e.second;
            else if (all_digits(e.first)) {   // N=action,argument
                std::string act, arg;
                if (!split2(e.second, act, arg)) continue;  // malformed line skipped
                ScriptAction a;
                a.action = std::atoi(act.c_str());
                a.argument = std::atoi(arg.c_str());
                sc.actions.push_back(a);
            }
        }
        if (sc.actions.empty()) continue;
        sc_ix[id] = (int)d.scripts.size();
        d.scripts.push_back(std::move(sc));
    }

    // [TeamTypes] index section: N=id [RE-GATE APPLIED: ai-team.md §1 — the
    // named-key set]. A team whose TaskForce/Script was dropped is dropped too.
    for (const auto& kv : ini.section("TeamTypes")) {
        const std::string id = upper(kv.second);
        if (id.empty() || !ini.has(id)) continue;
        TeamType t;
        t.id = id;
        t.name = ini.get(id, "Name");
        t.house = ini.get(id, "House");
        const auto tf = tf_ix.find(upper(ini.get(id, "TaskForce")));
        const auto sc = sc_ix.find(upper(ini.get(id, "Script")));
        if (tf == tf_ix.end() || sc == sc_ix.end()) continue;  // art-guard-drop
        t.taskforce = tf->second;
        t.script = sc->second;
        t.priority = ini.get_int(id, "Priority", 0);
        t.max = ini.get_int(id, "Max", 0);
        t.full = truthy(ini.get(id, "Full"));
        t.autocreate = truthy(ini.get(id, "Autocreate"));
        t.waypoint = waypoint_from_letter(ini.get(id, "Waypoint"));
        d.teams.push_back(std::move(t));
    }
    return d;
}
```

```cpp
namespace {
// Case-insensitive house-token match; "<none>"/"" matches any house
// [RE-GATE APPLIED: ai-team.md §4 — pre-hunt matching].
bool house_matches(const std::string& team_house, const std::string& house_name) {
    if (team_house.empty()) return true;
    std::string a = team_house, b = house_name;
    for (char& c : a) c = (char)std::toupper((unsigned char)c);
    for (char& c : b) c = (char)std::toupper((unsigned char)c);
    return a == "<NONE>" || a == b;
}
}  // namespace

int select_autocreate_team(const AimdData& aimd, const std::string& house_name,
                           const std::vector<TeamInstance>& live, double credits,
                           const AiTeamRules& rules) {
    // [RE: ai-team.md §4] autocreate ONLY (AITriggerTypes creation is AI-3);
    // house match; Max live-instance cap (Max<=0 -> cap 1 [RE-GATE §4]);
    // CreditReserve floor [RE-GATE §4/§6]; highest Priority wins, tie =
    // lowest TeamType index (deterministic, no RNG).
    if (credits < rules.credit_reserve) return -1;
    int best = -1;
    for (int i = 0; i < (int)aimd.teams.size(); ++i) {
        const TeamType& tt = aimd.teams[i];
        if (!tt.autocreate || !house_matches(tt.house, house_name)) continue;
        const int cap = tt.max > 0 ? tt.max : 1;
        int count = 0;
        for (const TeamInstance& t : live)
            if (t.teamtype == i) ++count;
        if (count >= cap) continue;
        if (best < 0 || tt.priority > aimd.teams[best].priority) best = i;
    }
    return best;
}

int recruit_idle(TeamInstance& team, const AimdData& aimd,
                 const std::vector<IdleUnitView>& idle) {
    // [RE: ai-team.md §2] recruit-idle-first: TaskForce slots in FILE ORDER,
    // ascending idle-unit walk per slot (the caller supplies `idle` in
    // ascending unit-key order) — mirrors AI-1's deterministic
    // standing-building sync. Each idle unit claimed at most once per call.
    if (team.teamtype < 0 || team.teamtype >= (int)aimd.teams.size()) return 0;
    const TaskForce& tf = aimd.taskforces[aimd.teams[team.teamtype].taskforce];
    std::vector<char> taken(idle.size(), 0);
    int claimed = 0;
    for (const TaskForceMember& m : tf.members) {
        int have = 0;
        for (int t : team.member_types)
            if (t == m.unit) ++have;
        for (std::size_t i = 0; i < idle.size() && have < m.count; ++i) {
            if (taken[i] || idle[i].unit_type != m.unit) continue;
            taken[i] = 1;
            team.members.push_back(idle[i].unit);
            team.member_types.push_back(m.unit);
            ++have;
            ++claimed;
        }
    }
    return claimed;
}

int next_produce_type(const std::vector<TeamInstance>& teams, const AimdData& aimd,
                      double credits, double cost_mul,
                      const std::vector<UnitTypeDef>& unit_roster,
                      const AiTeamRules& rules) {
    // [RE: ai-team.md §6] produce-the-shortfall: the FIRST (oldest)
    // Recruiting team's first unfilled slot drives production — one intent
    // per house tick; the caller routes infantry->tab2, vehicle->tab3 and
    // holds when that queue is busy (the single-slot queue IS the recovered
    // one-unit-at-a-time factory gate). Affordability gates the START:
    // credits >= cost x cost_mul + CreditReserve — bank, never substitute
    // (the AI-1 rule). InfantryReserve is recorded in `rules` but pre-hunt
    // does NOT gate team shortfall [RE-GATE APPLIED: ai-team.md §6 — amend
    // here if the hunt shows otherwise].
    for (const TeamInstance& team : teams) {
        if (team.state != TeamState::Recruiting) continue;
        const int t = team_shortfall_type(team, aimd);
        if (t < 0 || t >= (int)unit_roster.size()) continue;
        if (credits < unit_roster[t].cost * cost_mul + rules.credit_reserve)
            return -1;  // bank income; NEVER a cheaper substitute
        return t;
    }
    return -1;
}

double unit_build_speed_mul(const AiTeamRules& rules) {
    // BuildSlowdown consumed as a unit build-TIME multiplier [RE-GATE
    // APPLIED: ai-team.md §6 — kAiBuildSlowdownTimeMul is a pre-hunt
    // PLACEHOLDER the hunt must pin].
    return rules.build_slowdown ? kAiBuildSlowdownTimeMul : 1.0;
}
```

```cpp
std::vector<IdleUnitView> World::ai_idle_units(int house, const HouseAiRec& rec) const {
    // Recruitable = own, alive, non-dying, NOT a harvester, at rest, no
    // attack pursuit, typed in the unit roster, and not already claimed by
    // any team. ASCENDING unit index — the deterministic claim order
    // [RE: ai-team.md §2].
    std::vector<IdleUnitView> out;
    for (int i = 0; i < (int)units_.size(); ++i) {
        const Unit& u = units_[i];
        if (!u.alive || u.die_time >= 0.0 || u.house != house || u.is_harvester)
            continue;
        if (u.moving() || u.attack_order) continue;
        const int t = ai_unit_type_of(rec, u);

bool World::ai_quarry_exists(int house, int quarry) const {
    for (const Unit& u : units_)
        if (u.alive && u.die_time < 0.0 && u.house != house &&
            quarry_unit_match(quarry, u))
            return true;
    if (quarry_wants_buildings(quarry))
        for (const Building& b : buildings_)
            if (b.alive && b.state != Building::State::Dead && b.house != house)
                return true;
    return false;
}

int World::ai_pick_quarry_unit(int house, int quarry, int nx, int ny) const {
    // Nearest matching enemy unit by squared cell distance; tie -> lowest
    // index (strict <) [RE-GATE APPLIED: ai-team.md §5 — pre-hunt rule].
    int best = -1;
    long long bestd = 0;
    for (int i = 0; i < (int)units_.size(); ++i) {
        const Unit& u = units_[i];
        if (!u.alive || u.die_time >= 0.0 || u.house == house) continue;
        if (!quarry_unit_match(quarry, u)) continue;
        const long long dx = u.cell_x - nx, dy = u.cell_y - ny;
        const long long dd = dx * dx + dy * dy;
        if (best < 0 || dd < bestd) { best = i; bestd = dd; }
    }
    return best;
}

int World::ai_pick_quarry_building(int house, int quarry, int nx, int ny) const {
    if (!quarry_wants_buildings(quarry)) return -1;
    int best = -1;
    long long bestd = 0;
    for (int i = 0; i < (int)buildings_.size(); ++i) {
        const Building& b = buildings_[i];
        if (!b.alive || b.state == Building::State::Dead || b.house == house) continue;
        const long long dx = b.cx - nx, dy = b.cy - ny;
        const long long dd = dx * dx + dy * dy;
        if (best < 0 || dd < bestd) { best = i; bestd = dd; }
    }
    return best;
}

void World::ai_order_attack(int unit_idx, int target_idx) {

    // existing order_attack drives SELECTED house-0 units only; teams need

    if (unit_idx < 0 || unit_idx >= (int)units_.size()) return;
    if (target_idx < 0 || target_idx >= (int)units_.size()) return;
    Unit& u = units_[unit_idx];

    if (!u.alive || u.die_time >= 0.0) return;
    if (!tgt.alive || tgt.die_time >= 0.0 || tgt.house == u.house) return;
    u.target = target_idx;
    u.target_building = -1;   // mutually exclusive, as in order_attack
    u.attack_order = true;

}

void World::ai_execute_team_command(int house, HouseAiRec& rec, TeamInstance& team,
                                    const TeamCommand& c) {
    switch (c.kind) {
        case TeamCommand::Kind::MoveTeam: {
            // Deterministic per-member spread (ring offsets) so members do
            // not stack one cell; scatter widens the ring [RE-FALLBACK:
            // ai-team.md deviations — engine formation stand-in;
            // order_unit_move cancels cleanly if a spread cell is
            // unreachable].
            static const int dx[9] = {0, 1, -1, 0, 0, 1, -1, 1, -1};
            static const int dy[9] = {0, 0, 0, 1, -1, 1, -1, -1, 1};
            const int spread = c.scatter ? 3 : 1;
            for (std::size_t k = 0; k < team.members.size(); ++k) {
                const int m = team.members[k];
                if (m < 0 || m >= (int)units_.size()) continue;
                const Unit& u = units_[m];
                if (!u.alive || u.die_time >= 0.0) continue;
                order_unit_move(m, c.x + dx[k % 9] * spread, c.y + dy[k % 9] * spread);
            }
            break;
        }
        case TeamCommand::Kind::AttackQuarry: {
            for (int m : team.members) {
                if (m < 0 || m >= (int)units_.size()) continue;
                Unit& u = units_[m];
                if (!u.alive || u.die_time >= 0.0) continue;
                if (u.attack_order && (u.target >= 0 || u.target_building >= 0))
                    continue;  // already pursuing — no target thrash
                const int tu = ai_pick_quarry_unit(house, c.quarry, u.cell_x, u.cell_y);
                if (tu >= 0) { ai_order_attack(m, tu); continue; }
                const int tb =
                    ai_pick_quarry_building(house, c.quarry, u.cell_x, u.cell_y);
                if (tb >= 0) ai_order_attack_building(m, tb);
            }
            break;
        }
        case TeamCommand::Kind::GuardArea: {
            // Guard anchors only — the EXISTING step_combat auto-acquire and
            // return-to-anchor machinery does the rest [RE: ai-team.md §3].
            for (int m : team.members) {
                if (m < 0 || m >= (int)units_.size()) continue;
                Unit& u = units_[m];
                if (!u.alive || u.die_time >= 0.0) continue;
                u.guard_x = c.x;
                u.guard_y = c.y;
            }
            break;
        }
        case TeamCommand::Kind::Inert:  // logged skip — pc already advanced
        case TeamCommand::Kind::Wait:
        case TeamCommand::Kind::None:
        default:
            break;
    }
}

bool World::ai_launch_unit_ready(int house, HouseAiRec& rec, QueueCat cat) {
    // Exit handoff via the EXISTING primary_factory + launch_unit path (the
    // player's run_interactive twin) [RE: ai-team.md §6]. Returns true if the
    // ready slot was consumed.
    if (!rec.queues.ready(cat)) return false;
    const AiTeamConfig& tc = rec.cfg.team;
    const std::vector<int>& ft =
        cat == QueueCat::Infantry ? tc.inf_factory_types : tc.veh_factory_types;
    const int fi = primary_factory(buildings_, ft, house);
    if (fi < 0) return false;  // no factory: the lost-check cancels the queue
    const int uslot = rec.queues.producing(cat);
    if (uslot < 0 || uslot >= (int)tc.unit_roster.size()) {
        rec.queues.placed(cat);  // defensive: orphaned ready slot
        return true;
    }
    const BuildingTypeDef& fdef = rec.cfg.roster[buildings_[fi].type];
    const int uidx = launch_unit(fi, fdef, tc.unit_roster[uslot].combat);
    if (uidx < 0) return false;  // exit blocked -> HOLD ready (retry next frame)
    // Claim into the FIRST (oldest) Recruiting team owing this type —
    // deterministic; an unclaimed unit stays idle and recruitable
    // [RE: ai-team.md §2].
    for (TeamInstance& t : rec.team_state.teams) {
        if (t.state != TeamState::Recruiting) continue;
        if (team_shortfall_type(t, tc.aimd) == uslot) {
            t.members.push_back(uidx);
            t.member_types.push_back(uslot);
            break;
        }
    }
    rec.queues.placed(cat);
    return true;
}

void World::step_ai_teams(double dt) {
    for (int h = 0; h < (int)ai_houses_.size(); ++h) {
        HouseAiRec& rec = ai_houses_[h];
        if (!rec.is_ai || !rec.cfg.team.enabled) continue;  // FIREWALL (spec §5)
        const AiTeamConfig& tc = rec.cfg.team;
        AiTeamState& st = rec.team_state;

        // 1) Member-death sweep [RE: ai-team.md §2]: prune dead/dying members
        //    (parallel vectors stay parallel); the pure lifecycle then sees
        //    the reopened slot (full=yes teams drop back to Recruiting).
        for (TeamInstance& t : st.teams) {
            for (int k = (int)t.members.size() - 1; k >= 0; --k) {
                const int m = t.members[k];
                if (m < 0 || m >= (int)units_.size() || !units_[m].alive ||
                    units_[m].die_time >= 0.0) {
                    t.members.erase(t.members.begin() + k);
                    t.member_types.erase(t.member_types.begin() + k);
                }
            }
        }

        // 2) Factory-lost cancel for the unit queues (the queues themselves
        //    already TICKED inside step_ai_houses's rec.queues.tick — no new
        //    production code) [RE: ai-team.md §6].
        if (rec.queues.producing(QueueCat::Infantry) != -1 &&
            factory_category_lost(tc.inf_factory_types, h))
            rec.queues.cancel(QueueCat::Infantry);
        if (rec.queues.producing(QueueCat::Vehicle) != -1 &&
            factory_category_lost(tc.veh_factory_types, h))
            rec.queues.cancel(QueueCat::Vehicle);
        rec.credits = rec.queues.credits();

        // 3) Launch READY unit slots (Infantry then Vehicle — deterministic).
        ai_launch_unit_ready(h, rec, QueueCat::Infantry);
        ai_launch_unit_ready(h, rec, QueueCat::Vehicle);

        // 4) Autocreate clock [RE-GATE APPLIED: ai-team.md §4 — TeamDelays
        //    cadence]: at most ONE new team per gate tick, highest priority
        //    first, deterministic.
        st.team_clock += dt;
        if (st.team_clock >= st.team_gate) {
            st.team_clock = 0.0;
            st.team_gate = (double)tc.rules.team_delay_frames / 15.0;
            const int pick = select_autocreate_team(tc.aimd, tc.house_name,
                                                    st.teams, rec.credits, tc.rules);
            if (pick >= 0) {
                TeamInstance t;
                t.teamtype = pick;
                st.teams.push_back(t);
                ++st.teams_created;
            }
        }

        // 5) Recruit-idle first [RE: ai-team.md §2] (re-scanned per team so a
        //    unit claimed by an earlier team is not offered to a later one),
        //    then ONE production intent per house tick through the EXISTING
        //    queues — tab2 (Infantry) / tab3 (Vehicle) routing [RE: §6].
        for (TeamInstance& t : st.teams) {
            if (t.state != TeamState::Recruiting) continue;
            recruit_idle(t, tc.aimd, ai_idle_units(h, rec));
        }
        const int want = next_produce_type(st.teams, tc.aimd, rec.credits,
                                           rec.cfg.mods.cost_mul, tc.unit_roster,
                                           tc.rules);
        if (want >= 0) {
            const bool veh = tc.unit_roster[want].is_vehicle;
            const QueueCat cat = veh ? QueueCat::Vehicle : QueueCat::Infantry;
            const std::vector<int>& ft = veh ? tc.veh_factory_types : tc.inf_factory_types;
            const bool factory = !factory_category_lost(ft, h);
            if (rec.queues.producing(cat) == -1 && factory &&
                can_start_unit(rec.queues.state(cat), want, tc.unit_roster,
                               buildings_, rec.cfg.roster, h, rec.cfg.groups,
                               factory)) {
                // Scalar start + slot restamp — the AI-1 building idiom:
                // difficulty cost handicap on the charge; BuildTime handicap
                // AND the CONSUMED BuildSlowdown penalty on the speed
                // [RE-GATE APPLIED: ai-team.md §6].
                const int cost = tc.unit_roster[want].cost;
                const int charge = std::max(1, (int)(cost * rec.cfg.mods.cost_mul));
                rec.queues.set_credits(rec.credits);
                start(rec.queues.state(cat), cost, charge,
                      rec.cfg.build_speed * rec.cfg.mods.build_time_mul *
                          unit_build_speed_mul(tc.rules));
                rec.queues.state(cat).slot.type = want;
                rec.credits = rec.queues.credits();
            }
        }

        // 6) Step every team (pure brain) + execute its command (impure
        //    glue); Disbanding teams release survivors (they simply stop
        //    being members -> idle, re-recruitable) and are erased
        //    [RE: ai-team.md §2].
        for (int i = 0; i < (int)st.teams.size();) {
            TeamInstance& t = st.teams[i];
            TeamView view;
            ai_fill_team_view(h, rec, t, view);
            const TeamCommand c = step_team(t, view, tc.aimd, st.rng, dt);
            ai_execute_team_command(h, rec, t, c);
            if (t.state == TeamState::Disbanding)
                st.teams.erase(st.teams.begin() + i);
            else
                ++i;
        }
    }
}
```

```cpp
// FNV hash of the AI-2 end state for --demo-ai-team's two-run byte-identity
// gate [RE: ai-team.md]: the SORTED live-team rows {teamtype, state, pc,
// member count} + each team's SORTED member {cell, hp} rows + the
// produced-unit list (house units: image chars, cell, hp, alive) + credits +
// teams_created. A NEW function; world_hash/econ_hash/prod_hash/ai_base_hash
// UNTOUCHED.
uint64_t ai_team_hash(const ra2engine::World& world, int house) {
    uint64_t h = 1469598103934665603ULL;  // FNV offset basis
    if (const ra2engine::AiTeamState* st = world.ai_team_state(house)) {
        h = fnv_mix(h, static_cast<uint64_t>(st->teams_created));
        std::vector<std::array<int, 4>> rows;
        for (const ra2engine::TeamInstance& t : st->teams)
            rows.push_back({t.teamtype, (int)t.state, t.script_pc, (int)t.members.size()});
        std::sort(rows.begin(), rows.end());
        for (const auto& r : rows)
            for (int x : r) h = fnv_mix(h, static_cast<uint64_t>(x));
        for (const ra2engine::TeamInstance& t : st->teams) {
            std::vector<std::array<int, 3>> mem;
            for (int m : t.members) {
                if (m < 0 || m >= (int)world.units().size()) continue;
                const ra2engine::Unit& u = world.units()[m];
                mem.push_back({u.cell_x, u.cell_y, u.hp});
            }
            std::sort(mem.begin(), mem.end());
            for (const auto& r : mem)
                for (int x : r) h = fnv_mix(h, static_cast<uint64_t>(x));
        }
    }
    for (const ra2engine::Unit& u : world.units()) {   // produced-unit list
        if (u.house != house) continue;
        for (char ch : u.image) h = fnv_mix(h, static_cast<uint64_t>((unsigned char)ch));
        h = fnv_mix(h, static_cast<uint64_t>(u.cell_x));
        h = fnv_mix(h, static_cast<uint64_t>(u.cell_y));
        h = fnv_mix(h, static_cast<uint64_t>(u.hp));
        h = fnv_mix(h, static_cast<uint64_t>(u.alive ? 1 : 0));
    }
    h = fnv_mix(h, static_cast<uint64_t>(std::llround(world.ai_credits(house))));
    return h;
}
```
