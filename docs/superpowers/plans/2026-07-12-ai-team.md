```cpp
#include "aimd_parse.hpp"
#include <cctype>
#include <cstdlib>
#include <unordered_map>

namespace ra2engine {
namespace {

std::string upper(std::string s) {
    for (char& c : s) c = (char)std::toupper((unsigned char)c);
    return s;
}
// RA2 boolean values: yes/no (also true/1 tolerated) [RE: ai-team.md §1].
bool truthy(const std::string& v) {
    if (v.empty()) return false;
    const char c = (char)std::toupper((unsigned char)v[0]);
    return c == 'Y' || c == 'T' || c == '1';
}
bool all_digits(const std::string& s) {
    if (s.empty()) return false;
    for (char c : s)
        if (!std::isdigit((unsigned char)c)) return false;
    return true;
}
// "count,UNITID" / "action,argument": split on the FIRST comma.
bool split2(const std::string& v, std::string& a, std::string& b) {
    const std::size_t comma = v.find(',');
    if (comma == std::string::npos) return false;
    a = v.substr(0, comma);
    b = v.substr(comma + 1);
    return true;
}

}  // namespace

int waypoint_from_letter(const std::string& s) {
    // Bijective base-26: A=0..Z=25, AA=26 [RE-GATE APPLIED: ai-team.md §1].
    if (s.empty()) return -1;
    long v = 0;
    for (char c : s) {
        const char u = (char)std::toupper((unsigned char)c);
        if (u < 'A' || u > 'Z') return -1;
        v = v * 26 + (u - 'A' + 1);
    }
    return (int)(v - 1);
}

AimdData parse_aimd(const ra2assets::Ini& ini,
                    const std::function<int(const std::string&)>& unit_index) {
    AimdData d;
    std::unordered_map<std::string, int> tf_ix, sc_ix;  // entity id -> table index

    // [TaskForces] index section: N=id, file order [RE: ai-team.md §1].
    for (const auto& kv : ini.section("TaskForces")) {
        const std::string id = upper(kv.second);
        if (id.empty() || !ini.has(id) || tf_ix.count(id)) continue;  // dangling/dup
        TaskForce tf;
        tf.id = id;
        bool ok = true;
        for (const auto& e : ini.section(id)) {
            if (e.first == "NAME") tf.name = e.second;
            else if (e.first == "GROUP") tf.group = std::atoi(e.second.c_str());
            else if (all_digits(e.first)) {   // N=count,UNITID
                std::string cnt, uid;
                if (!split2(e.second, cnt, uid)) { ok = false; break; }
                TaskForceMember m;
                m.count = std::atoi(cnt.c_str());
                m.unit = unit_index(upper(uid));
                if (m.count <= 0 || m.unit < 0) { ok = false; break; }  // art-guard-drop
                tf.members.push_back(m);
            }
        }
        if (!ok || tf.members.empty()) continue;  // whole TaskForce dropped
        tf_ix[id] = (int)d.taskforces.size();
        d.taskforces.push_back(std::move(tf));
    }

    // [ScriptTypes] index section: N=id [RE: ai-team.md §1/§3]. Every action is
    // KEPT even if this engine interprets it inert — the inert split is the
    // INTERPRETER's job (ai_team), not the parser's.
    for (const auto& kv : ini.section("ScriptTypes")) {
        const std::string id = upper(kv.second);
        if (id.empty() || !ini.has(id) || sc_ix.count(id)) continue;
        ScriptType sc;
        sc.id = id;
        for (const auto& e : ini.section(id)) {
            if (e.first == "NAME") sc.name = e.second;
            else if (all_digits(e.first)) {   // N=action,argument
                std::string act, arg;
                if (!split2(e.second, act, arg)) continue;  // malformed line skipped
                ScriptAction a;
                a.action = std::atoi(act.c_str());
                a.argument = std::atoi(arg.c_str());
                sc.actions.push_back(a);
            }
        }
        if (sc.actions.empty()) continue;
        sc_ix[id] = (int)d.scripts.size();
        d.scripts.push_back(std::move(sc));
    }

    // [TeamTypes] index section: N=id [RE-GATE APPLIED: ai-team.md §1 — the
    // named-key set]. A team whose TaskForce/Script was dropped is dropped too.
    for (const auto& kv : ini.section("TeamTypes")) {
        const std::string id = upper(kv.second);
        if (id.empty() || !ini.has(id)) continue;
        TeamType t;
        t.id = id;
        t.name = ini.get(id, "Name");
        t.house = ini.get(id, "House");
        const auto tf = tf_ix.find(upper(ini.get(id, "TaskForce")));
        const auto sc = sc_ix.find(upper(ini.get(id, "Script")));
        if (tf == tf_ix.end() || sc == sc_ix.end()) continue;  // art-guard-drop
        t.taskforce = tf->second;
        t.script = sc->second;
        t.priority = ini.get_int(id, "Priority", 0);
        t.max = ini.get_int(id, "Max", 0);
        t.full = truthy(ini.get(id, "Full"));
        t.autocreate = truthy(ini.get(id, "Autocreate"));
        t.aggressive = truthy(ini.get(id, "Aggressive"));   // [RE: ai-team.md §1]
        t.reinforce = truthy(ini.get(id, "Reinforce"));
        t.suicide = truthy(ini.get(id, "Suicide"));
        t.tech_level = ini.get_int(id, "TechLevel", 0);
        t.is_base_defense = truthy(ini.get(id, "IsBaseDefense"));  // DEFENSIVE marker
        t.waypoint = waypoint_from_letter(ini.get(id, "Waypoint"));
        d.teams.push_back(std::move(t));
    }
    return d;
}

}  // namespace ra2engine
```

```cpp
OpClass classify_op(int action) {
    // [RE-GATE APPLIED: ai-team.md §3 — the FULL recovered dispatch table. The
    // classifier FUN_006e30d0 enumerates ~90 valid actions (up to 0x90) via a
    // jump table; this switch is the single table location. The IMPLEMENTABLE
    // core below is exactly the recovered set; EVERY other valid action
    // (transport 0x2e/0x2f/0x50/0x6b, aircraft 0x1e/0x55-0x57, superweapon/
    // support 0x3d/0x3e, droppod, cosmetics) and every unrecognised number is
    // Inert: logged skip, pc++ — parsed + RE-documented, never executed
    // (spec §2 scope bound).]
    switch (action) {
        case kOpAttackQuarry:     return OpClass::AttackQuarry;
        case kOpAttackMove:       return OpClass::AttackQuarry;  // 0x3f = quarry + regroup
        case kOpAttackHouse:      return OpClass::AttackHouse;
        case kOpAttackBuilding:   return OpClass::AttackBuilding;
        case kOpAttackUnits:      return OpClass::AttackUnits;
        case kOpMoveToWaypoint:   return OpClass::MoveWaypoint;
        case kOpMoveToCell:       return OpClass::MoveCell;
        case kOpGuardArea:        return OpClass::GuardArea;
        case kOpTimedGuard:       return OpClass::TimedGuard;
        case kOpTimedGuardAlt:    return OpClass::TimedGuard;    // 0x3a same unit as 5
        case kOpRegroup:          return OpClass::Regroup;
        case kOpScatter:          return OpClass::Scatter;
        case kOpJumpToLine:       return OpClass::Jump;
        case kOpDisband:          return OpClass::Disband;

Cellxy gather_cell(int ex, int ey, std::uint32_t& rng) {
    // [RE-GATE APPLIED: ai-team.md §2] — a point on a circle of radius
    // kAiSafeDistance (AISafeDistance = 20 cells) around the ENEMY base centre
    // (ex,ey), at a SEEDED-xorshift32 angle from `rng` (ai_base's ai_rand
    // substituting the lockstep RNG [DEVIATION]). The caller snaps this ideal
    // ring cell to the nearest free cell via the existing move/path machinery.
    const double theta =
        (double)ai_rand(rng) / 4294967296.0 * 6.283185307179586;  // [0, 2*pi)
    const double rad = (double)kAiSafeDistance;
    return Cellxy{ex + (int)std::lround(rad * std::cos(theta)),
                  ey + (int)std::lround(rad * std::sin(theta))};
}

namespace {
// Alive-member helpers over the PARALLEL view.
int alive_count(const TeamView& v) {
    int n = 0;
    for (const TeamMemberView& m : v.members)
        if (m.alive) ++n;
    return n;
}
// All live members within `cells` of (cx,cy) — the recovered Stray/CloseEnough
// radius predicate [RE-GATE APPLIED: ai-team.md §2]. Squared-distance in cells.
bool all_within(const TeamView& v, int cx, int cy, double cells) {
    const double r2 = cells * cells;
    for (const TeamMemberView& m : v.members) {
        if (!m.alive) continue;
        const double dx = (double)(m.cx - cx), dy = (double)(m.cy - cy);
        if (dx * dx + dy * dy > r2) return false;
    }
    return true;
}
Cellxy centroid(const TeamView& v) {
    long long sx = 0, sy = 0, n = 0;
    for (const TeamMemberView& m : v.members)
        if (m.alive) { sx += m.cx; sy += m.cy; ++n; }
    if (n == 0) return Cellxy{v.base_x, v.base_y};
    return Cellxy{(int)(sx / n), (int)(sy / n)};
}

TeamCommand step_team(TeamInstance& team, const TeamView& view,
                      const AimdData& aimd, std::uint32_t& rng, double dt,
                      bool multiplay) {
    TeamCommand c;
    if (team.teamtype < 0 || team.teamtype >= (int)aimd.teams.size()) {
        team.state = TeamState::Disbanding;
        return c;
    }
    const TeamType& tt = aimd.teams[team.teamtype];
    const ScriptType& sc = aimd.scripts[tt.script];
    const int alive = alive_count(view);

    // Team wiped after launch -> Disbanding, any state [RE: ai-team.md §2].
    if (team.state != TeamState::Recruiting && alive == 0) {
        team.state = TeamState::Disbanding;
        return c;
    }

    switch (team.state) {
        case TeamState::Recruiting: {
            team.recruit_timer += dt;
            const bool complete = team_shortfall_type(team, aimd) < 0;
            // Full=yes: WAIT for full TaskForce strength before leaving
            // Recruiting. Full=no (all stock offensive teams): ACT UNDER
            // STRENGTH — launch as soon as it has any member, never waiting for
            // the dissolve timer [RE-GATE APPLIED: ai-team.md §2].
            if (complete || (!tt.full && alive > 0)) {
                team.state = TeamState::Gathering;
                const Cellxy g = gather_cell(view.enemy_x, view.enemy_y, rng);
                team.gather_x = g.x;
                team.gather_y = g.y;
                team.move_issued = false;
                return c;
            }
            // DissolveUnfilledTeamDelay is MULTIPLAY ONLY: only an MP unfilled
            // team dissolves on the timer. The skirmish/campaign default
            // (multiplay=false) NEVER dissolves — it keeps recruiting/producing
            // toward strength [RE-GATE APPLIED: ai-team.md §2].
            if (multiplay &&
                team.recruit_timer * 15.0 >= (double)kAiDissolveUnfilledDelayFrames)
                team.state = TeamState::Disbanding;
            return c;  // Recruiting emits no field command (caller recruits/produces)
        }

        case TeamState::Gathering: {
            // A death that reopens a slot drops a full=yes team back to
            // Recruiting [RE: ai-team.md §2].
            if (tt.full && team_shortfall_type(team, aimd) >= 0) {
                team.state = TeamState::Recruiting;
                team.recruit_timer = 0.0;
                team.move_issued = false;
                return c;
            }
            if (!team.move_issued) {
                c.kind = TeamCommand::Kind::MoveTeam;
                c.x = team.gather_x;
                c.y = team.gather_y;
                team.move_issued = true;
                return c;
            }
            // GATHERED = all live members within Rules.Stray (2.0 cells) of the
            // gather focus [RE-GATE APPLIED: ai-team.md §2].
            if (all_within(view, team.gather_x, team.gather_y, kAiStrayCells)) {
                team.state = TeamState::Executing;
                team.script_pc = 0;
                team.script_timer = 0.0;
                team.move_issued = false;
            }
            return c;
        }

        case TeamState::Executing: {
            if (team.script_pc < 0 || team.script_pc >= (int)sc.actions.size()) {
                team.state = TeamState::Disbanding;  // script ran off the end
                return c;
            }
            const ScriptAction a = sc.actions[team.script_pc];
            const OpClass oc = classify_op(a.action);

            // The four attack classes share ONE emit-while-target-exists /
            // advance-when-extinct body, differing only in the quarry class they
            // resolve. The caller (World) computes view.quarry_exists for THIS
            // action's effective quarry [RE-GATE APPLIED: ai-team.md §3/§5].
            if (oc == OpClass::AttackQuarry || oc == OpClass::AttackHouse ||
                oc == OpClass::AttackBuilding || oc == OpClass::AttackUnits) {
                if (!view.quarry_exists) {
                    ++team.script_pc;  // target class extinct -> advance
                    team.move_issued = false;
                    return c;
                }
                c.kind = TeamCommand::Kind::AttackQuarry;
                c.quarry = (oc == OpClass::AttackQuarry) ? a.argument
                         : (oc == OpClass::AttackBuilding) ? kQuarryBuildings
                         : kQuarryAnything;  // AttackHouse / AttackUnits (§5)
                return c;                    // re-emitted until the class is gone
            }

            switch (oc) {
                case OpClass::MoveWaypoint: {
                    const Cellxy w =
                        (a.argument >= 0 && a.argument < (int)view.waypoints.size())
                            ? view.waypoints[a.argument]
                            : Cellxy{view.enemy_x, view.enemy_y};  // [RE-FALLBACK:
                              // unmapped waypoint -> enemy base centre]
                    if (!team.move_issued) {
                        c.kind = TeamCommand::Kind::MoveTeam;
                        c.x = w.x;
                        c.y = w.y;
                        team.move_issued = true;
                    } else if (all_within(view, w.x, w.y, kAiCloseEnoughCells)) {
                        ++team.script_pc;          // arrived (CloseEnough) -> next
                        team.move_issued = false;
                    }
                    return c;
                }
                case OpClass::MoveCell: {
                    // [RE-FALLBACK: ai-team.md §3 — argument is a PACKED cell; the
                    // pure module has no map width to unpack it, so it resolves to
                    // the enemy base centre (march-at-enemy) until packed-cell
                    // decoding lands (AI-3).]
                    const Cellxy w{view.enemy_x, view.enemy_y};
                    if (!team.move_issued) {
                        c.kind = TeamCommand::Kind::MoveTeam;
                        c.x = w.x;
                        c.y = w.y;
                        team.move_issued = true;
                    } else if (all_within(view, w.x, w.y, kAiCloseEnoughCells)) {
                        ++team.script_pc;
                        team.move_issued = false;
                    }
                    return c;
                }
                case OpClass::Regroup: {
                    const Cellxy m = centroid(view);
                    if (!team.move_issued) {
                        c.kind = TeamCommand::Kind::MoveTeam;
                        c.x = m.x;
                        c.y = m.y;
                        team.move_issued = true;
                    } else if (all_within(view, m.x, m.y, kAiCloseEnoughCells)) {
                        ++team.script_pc;
                        team.move_issued = false;
                    }
                    return c;
                }
                case OpClass::GuardArea: {
                    // 0x0c guard area at current location: set the guard anchors
                    // then advance (the EXISTING auto-acquire / return-to-anchor
                    // machinery does the rest). One-shot so an untimed guard can
                    // never wedge the script [RE-GATE APPLIED: ai-team.md §3].
                    const Cellxy m = centroid(view);
                    c.kind = TeamCommand::Kind::GuardArea;
                    c.x = m.x;
                    c.y = m.y;
                    c.radius = kAiGatherRadius;
                    ++team.script_pc;
                    team.move_issued = false;
                    return c;
                }
                case OpClass::TimedGuard: {
                    // 0x05 / 0x3a guard for <arg> tenths-of-min; arg x 90 frames
                    // at 15 Hz [RE-GATE APPLIED: ai-team.md §3].
                    team.script_timer += dt * 15.0;  // accumulate FRAMES
                    if (team.script_timer >=
                        (double)(a.argument * kAiGuardTicksToFrames)) {
                        ++team.script_pc;
                        team.script_timer = 0.0;
                        team.move_issued = false;
                        return c;
                    }
                    const Cellxy m = centroid(view);
                    c.kind = TeamCommand::Kind::GuardArea;
                    c.x = m.x;
                    c.y = m.y;
                    c.radius = kAiGatherRadius;
                    return c;
                }
                case OpClass::Jump: {
                    // 0x06 Jump consumes the tick — a jump cycle can never spin
                    // within one tick; argument is the 0-based script line index
                    // [RE-GATE APPLIED: ai-team.md §3 — CONFIRMED].
                    team.script_pc =
                        std::max(0, std::min(a.argument, (int)sc.actions.size()));
                    team.script_timer = 0.0;
                    team.move_issued = false;
                    return c;
                }
                case OpClass::Scatter: {
                    const Cellxy m = centroid(view);
                    c.kind = TeamCommand::Kind::MoveTeam;
                    c.x = m.x;
                    c.y = m.y;
                    c.scatter = true;  // caller spreads members [RE: ai-team.md §3]
                    ++team.script_pc;  // one-shot
                    team.move_issued = false;
                    return c;
                }
                case OpClass::Disband: {
                    // 0x31 (49): set success flag then disband [RE: ai-team.md §3].
                    team.state = TeamState::Disbanding;
                    return c;
                }
                case OpClass::Inert:
                default: {
                    // Valid-but-unimplementable / unrecognised opcode: logged
                    // skip, pc++ [RE-GATE APPLIED: ai-team.md §3; spec §2].
                    c.kind = TeamCommand::Kind::Inert;
                    c.opcode = a.action;
                    ++team.script_pc;
                    team.move_issued = false;
                    return c;
                }
            }
        }

        case TeamState::Disbanding:
        default:
            return c;  // caller releases survivors + erases the instance
    }
}
```

```cpp
namespace {
// Optional country restriction ONLY [RE-GATE APPLIED: ai-team.md §1/§4]: stock
// data is universally House=<none>, so this returns true for "<none>"/"" (any
// house) — it is NOT the eligibility gate (fielding the TaskForce is, guaranteed
// upstream by parse_aimd's art-guard-drop). A rare House=<name> team restricts
// to that country (case-insensitive).
bool house_matches(const std::string& team_house, const std::string& house_name) {
    if (team_house.empty()) return true;
    std::string a = team_house, b = house_name;
    for (char& c : a) c = (char)std::toupper((unsigned char)c);
    for (char& c : b) c = (char)std::toupper((unsigned char)c);
    return a == "<NONE>" || a == b;
}
}  // namespace

int select_autocreate_team(const AimdData& aimd, const std::string& house_name,
                           const std::vector<TeamInstance>& live, double credits,
                           const AiTeamRules& rules) {
    // [RE-GATE APPLIED: ai-team.md §4] autocreate ONLY (AITriggerTypes creation
    // is AI-3); NOT is_base_defense (defensive teams out of offensive scope);
    // the TotalAITeamCap hard per-house cap; the Max live-instance cap with
    // Max=0 = UNCAPPED; the CreditReserve spend floor; highest Priority wins,
    // tie = lowest TeamType index (deterministic). Eligibility is NOT a per-side
    // house-token compare — house_matches only applies a rare House=<name>
    // country restriction; fielding the TaskForce is guaranteed by parse_aimd's
    // art-guard-drop (a team in `aimd` has a TaskForce this house can field).
    if (credits < rules.credit_reserve) return -1;
    if ((int)live.size() >= rules.total_team_cap) return -1;  // hard per-house cap
    int best = -1;
    for (int i = 0; i < (int)aimd.teams.size(); ++i) {
        const TeamType& tt = aimd.teams[i];
        if (!tt.autocreate || tt.is_base_defense ||
            !house_matches(tt.house, house_name))
            continue;
        int count = 0;
        for (const TeamInstance& t : live)
            if (t.teamtype == i) ++count;
        if (tt.max > 0 && count >= tt.max) continue;  // Max=0 -> uncapped
        if (best < 0 || tt.priority > aimd.teams[best].priority) best = i;
    }
    return best;
}

int recruit_idle(TeamInstance& team, const AimdData& aimd,
                 const std::vector<IdleUnitView>& idle) {
    // [RE: ai-team.md §2] recruit-idle-first: TaskForce slots in FILE ORDER,
    // ascending idle-unit walk per slot (the caller supplies `idle` in
    // ascending unit-key order) — mirrors AI-1's deterministic
    // standing-building sync. Each idle unit claimed at most once per call.
    if (team.teamtype < 0 || team.teamtype >= (int)aimd.teams.size()) return 0;
    const TaskForce& tf = aimd.taskforces[aimd.teams[team.teamtype].taskforce];
    std::vector<char> taken(idle.size(), 0);
    int claimed = 0;
    for (const TaskForceMember& m : tf.members) {
        int have = 0;
        for (int t : team.member_types)
            if (t == m.unit) ++have;
        for (std::size_t i = 0; i < idle.size() && have < m.count; ++i) {
            if (taken[i] || idle[i].unit_type != m.unit) continue;
            taken[i] = 1;
            team.members.push_back(idle[i].unit);
            team.member_types.push_back(m.unit);
            ++have;
            ++claimed;
        }
    }
    return claimed;
}

int next_produce_type(const std::vector<TeamInstance>& teams, const AimdData& aimd,
                      double credits, double cost_mul,
                      const std::vector<UnitTypeDef>& unit_roster,
                      const AiTeamRules& rules) {
    // [RE: ai-team.md §6] produce-the-shortfall: the FIRST (oldest)
    // Recruiting team's first unfilled slot drives production — one intent
    // per house tick; the caller routes infantry->tab2, vehicle->tab3 and
    // holds when that queue is busy (the single-slot queue IS the recovered
    // one-unit-at-a-time factory gate). Affordability gates the START:
    // credits >= cost x cost_mul + CreditReserve — bank, never substitute
    // (the AI-1 rule). InfantryReserve is recorded in `rules` but pre-hunt
    // does NOT gate team shortfall [RE-GATE APPLIED: ai-team.md §6 — amend
    // here if the hunt shows otherwise].
    for (const TeamInstance& team : teams) {
        if (team.state != TeamState::Recruiting) continue;
        const int t = team_shortfall_type(team, aimd);
        if (t < 0 || t >= (int)unit_roster.size()) continue;
        // INFANTRY shortfall is produced FREELY above InfantryReserve (50000);
        // otherwise it falls to the normal affordability check. VEHICLE shortfall
        // always uses the normal check [RE-GATE APPLIED: ai-team.md §6].
        const bool infantry = !unit_roster[t].is_vehicle;
        if (infantry && credits > rules.infantry_reserve) return t;
        if (credits < unit_roster[t].cost * cost_mul + rules.credit_reserve)
            return -1;  // bank income; NEVER a cheaper substitute
        return t;
    }
    return -1;
}

bool build_slowdown_blocks(const AiTeamRules& rules, bool category_has_queued_or_onhand) {
    // BuildSlowdown is a difficulty BYTE (Easy off / Normal+Difficult on), an
    // anti-spam OVER-QUEUE damp, NOT a build-time multiplier [RE-GATE APPLIED:
    // ai-team.md §6]. When on, decline to START a category the house already has
    // queued or on-hand.
    return rules.build_slowdown && category_has_queued_or_onhand;
}
```

```cpp
std::vector<IdleUnitView> World::ai_idle_units(int house, const HouseAiRec& rec) const {
    // Recruitable = own, alive, non-dying, NOT a harvester, at rest, no
    // attack pursuit, typed in the unit roster, and not already claimed by
    // any team. ASCENDING unit index — the deterministic claim order
    // [RE: ai-team.md §2].
    std::vector<IdleUnitView> out;
    for (int i = 0; i < (int)units_.size(); ++i) {
        const Unit& u = units_[i];
        if (!u.alive || u.die_time >= 0.0 || u.house != house || u.is_harvester)
            continue;
        if (u.moving() || u.attack_order) continue;
        const int t = ai_unit_type_of(rec, u);
        if (t < 0) continue;
        bool in_team = false;
        for (const TeamInstance& tm : rec.team_state.teams) {
            for (int m : tm.members)
                if (m == i) { in_team = true; break; }
            if (in_team) break;
        }

int World::ai_pick_quarry_unit(int house, int quarry, int nx, int ny,
                               std::uint32_t& rng) const {
    // Class-filter + distance^2 x value x seeded-RNG jitter, keep the minimum
    // (nearest-and-most-valuable, with jitter); tie -> lowest index
    // [RE-GATE APPLIED: ai-team.md §5]. `value` is a per-target term (proxy 1
    // pre-map [RE-FALLBACK]); the jitter draws from ai_rand over `rng`.
    int best = -1;
    long long bestd = 0;
    for (int i = 0; i < (int)units_.size(); ++i) {
        const Unit& u = units_[i];
        if (!u.alive || u.die_time >= 0.0 || u.house == house) continue;
        if (!quarry_unit_match(quarry, u)) continue;
        const long long dx = u.cell_x - nx, dy = u.cell_y - ny;
        const long long value = 1;                      // [RE-FALLBACK] proxy
        const long long jitter = 90 + (ai_rand(rng) % 21);  // 0.90x .. 1.10x, x100
        const long long score = (dx * dx + dy * dy) * value * jitter;
        if (best < 0 || score < bestd) { best = i; bestd = score; }
    }
    return best;
}

int World::ai_pick_quarry_building(int house, int quarry, int nx, int ny,
                                   std::uint32_t& rng) const {
    if (!quarry_wants_buildings(quarry)) return -1;
    int best = -1;
    long long bestd = 0;
    for (int i = 0; i < (int)buildings_.size(); ++i) {
        const Building& b = buildings_[i];
        if (!b.alive || b.state == Building::State::Dead || b.house == house) continue;
        const long long dx = b.cx - nx, dy = b.cy - ny;
        const long long value = 1;                      // [RE-FALLBACK] proxy
        const long long jitter = 90 + (ai_rand(rng) % 21);
        const long long score = (dx * dx + dy * dy) * value * jitter;
        if (best < 0 || score < bestd) { best = i; bestd = score; }
    }
    return best;
}

    Unit& u = units_[unit_idx];

    u.target_building = target_building;
    u.attack_order = true;

void World::ai_order_attack(int unit_idx, int target_idx) {
    // Per-unit entry for AI-owned units: the existing order_attack drives
    // SELECTED house-0 units only; teams need per-unit orders for ANY house.
    // ADDITIVE — the player path is untouched (the six gates verify). Keeps its
    // OWN guard, then calls the shared commit_attack_target [RE: ai-team.md §5].
    if (unit_idx < 0 || unit_idx >= (int)units_.size()) return;
    if (target_idx < 0 || target_idx >= (int)units_.size()) return;
    const Unit& u = units_[unit_idx];
    const Unit& tgt = units_[target_idx];
    if (!u.alive || u.die_time >= 0.0) return;
    if (!tgt.alive || tgt.die_time >= 0.0 || tgt.house == u.house) return;  // AI guard
    commit_attack_target(unit_idx, target_idx, /*target_building=*/-1);
}

void World::ai_order_attack_building(int unit_idx, int b_idx) {
    // Per-unit entry (same rationale); own guard + shared commit_attack_target.
    if (unit_idx < 0 || unit_idx >= (int)units_.size()) return;
    if (b_idx < 0 || b_idx >= (int)buildings_.size()) return;
    const Unit& u = units_[unit_idx];
    const Building& tgt = buildings_[b_idx];
    if (!u.alive || u.die_time >= 0.0) return;
    if (!tgt.alive || tgt.state == Building::State::Dead || tgt.house == u.house) return;
    commit_attack_target(unit_idx, /*target_unit=*/-1, b_idx);
}

void World::ai_fill_team_view(int house, const HouseAiRec& rec,
                              const TeamInstance& team, TeamView& v) const {
    v.members.clear();
    for (int m : team.members) {
        TeamMemberView mv;
        mv.unit = m;
        if (m >= 0 && m < (int)units_.size()) {
            const Unit& u = units_[m];
            mv.alive = u.alive && u.die_time < 0.0;
            mv.cx = u.cell_x;
            mv.cy = u.cell_y;
            mv.idle = !u.moving() &&
                      !(u.attack_order && (u.target >= 0 || u.target_building >= 0));
        }
        v.members.push_back(mv);
    }
    v.waypoints = rec.cfg.team.waypoints;
    // Base center + enemy centroid reuse the AI-1 view (ConYard-first center,
    // rival-buildings threat centroid) — ONE source of truth.
    const BaseView bv = make_ai_base_view(house, rec);
    v.base_x = bv.center_x;
    v.base_y = bv.center_y;
    v.enemy_x = bv.threat_x;
    v.enemy_y = bv.threat_y;
    // quarry_exists is computed for the CURRENT attack action's EFFECTIVE quarry
    // class — the completion predicate the interpreter consults. It must match
    // the quarry class step_team sets on the emitted command [RE: ai-team.md
    // §3/§5]: AttackQuarry -> argument; AttackBuilding -> Buildings;
    // AttackHouse/AttackUnits -> Anything.
    v.quarry_exists = false;
    const AimdData& aimd = rec.cfg.team.aimd;
    if (team.state == TeamState::Executing && team.teamtype >= 0 &&
        team.teamtype < (int)aimd.teams.size()) {
        const ScriptType& sc = aimd.scripts[aimd.teams[team.teamtype].script];
        if (team.script_pc >= 0 && team.script_pc < (int)sc.actions.size()) {
            const ScriptAction a = sc.actions[team.script_pc];
            const OpClass oc = classify_op(a.action);
            int q = -1;
            if (oc == OpClass::AttackQuarry)        q = a.argument;
            else if (oc == OpClass::AttackBuilding) q = kQuarryBuildings;
            else if (oc == OpClass::AttackHouse ||
                     oc == OpClass::AttackUnits)    q = kQuarryAnything;
            if (q >= 0) v.quarry_exists = ai_quarry_exists(house, q);
        }
    }
}

void World::ai_execute_team_command(int house, HouseAiRec& rec, TeamInstance& team,
                                    const TeamCommand& c) {
    switch (c.kind) {
        case TeamCommand::Kind::MoveTeam: {
            // Deterministic per-member spread (ring offsets) so members do
            // not stack one cell; scatter widens the ring [RE-FALLBACK:
            // ai-team.md deviations — engine formation stand-in;
            // order_unit_move cancels cleanly if a spread cell is
            // unreachable].
            static const int dx[9] = {0, 1, -1, 0, 0, 1, -1, 1, -1};
            static const int dy[9] = {0, 0, 0, 1, -1, 1, -1, -1, 1};
            const int spread = c.scatter ? 3 : 1;
            for (std::size_t k = 0; k < team.members.size(); ++k) {
                const int m = team.members[k];
                if (m < 0 || m >= (int)units_.size()) continue;
                const Unit& u = units_[m];
                if (!u.alive || u.die_time >= 0.0) continue;
                order_unit_move(m, c.x + dx[k % 9] * spread, c.y + dy[k % 9] * spread);
            }
            break;
        }
        case TeamCommand::Kind::AttackQuarry: {
            for (int m : team.members) {
                if (m < 0 || m >= (int)units_.size()) continue;
                Unit& u = units_[m];
                if (!u.alive || u.die_time >= 0.0) continue;
                if (u.attack_order && (u.target >= 0 || u.target_building >= 0))
                    continue;  // already pursuing — no target thrash
                const int tu = ai_pick_quarry_unit(house, c.quarry, u.cell_x,
                                                   u.cell_y, rec.team_state.rng);
                if (tu >= 0) { ai_order_attack(m, tu); continue; }
                const int tb = ai_pick_quarry_building(house, c.quarry, u.cell_x,
                                                       u.cell_y, rec.team_state.rng);
                if (tb >= 0) ai_order_attack_building(m, tb);
            }
            break;
        }
        case TeamCommand::Kind::GuardArea: {
            // Guard anchors only — the EXISTING step_combat auto-acquire and
            // return-to-anchor machinery does the rest [RE: ai-team.md §3].
            for (int m : team.members) {
                if (m < 0 || m >= (int)units_.size()) continue;
                Unit& u = units_[m];
                if (!u.alive || u.die_time >= 0.0) continue;
                u.guard_x = c.x;
                u.guard_y = c.y;
            }
            break;
        }
        case TeamCommand::Kind::Inert:  // logged skip — pc already advanced
        case TeamCommand::Kind::Wait:
        case TeamCommand::Kind::None:
        default:
            break;
    }
}

bool World::ai_launch_unit_ready(int house, HouseAiRec& rec, QueueCat cat) {
    // Exit handoff via the EXISTING primary_factory + launch_unit path (the
    // player's run_interactive twin) [RE: ai-team.md §6]. Returns true if the
    // ready slot was consumed.
    if (!rec.queues.ready(cat)) return false;
    const AiTeamConfig& tc = rec.cfg.team;
    const std::vector<int>& ft =
        cat == QueueCat::Infantry ? tc.inf_factory_types : tc.veh_factory_types;
    const int fi = primary_factory(buildings_, ft, house);
    if (fi < 0) return false;  // no factory: the lost-check cancels the queue
    const int uslot = rec.queues.producing(cat);
    if (uslot < 0 || uslot >= (int)tc.unit_roster.size()) {
        rec.queues.placed(cat);  // defensive: orphaned ready slot
        return true;
    }
    const BuildingTypeDef& fdef = rec.cfg.roster[buildings_[fi].type];
    const int uidx = launch_unit(fi, fdef, tc.unit_roster[uslot].combat);
    if (uidx < 0) return false;  // exit blocked -> HOLD ready (retry next frame)
    // Claim into the FIRST (oldest) Recruiting team owing this type —
    // deterministic; an unclaimed unit stays idle and recruitable
    // [RE: ai-team.md §2].
    for (TeamInstance& t : rec.team_state.teams) {
        if (t.state != TeamState::Recruiting) continue;
        if (team_shortfall_type(t, tc.aimd) == uslot) {
            t.members.push_back(uidx);
            t.member_types.push_back(uslot);
            break;
        }
    }
    rec.queues.placed(cat);
    return true;
}

void World::step_ai_teams(double dt) {
    for (int h = 0; h < (int)ai_houses_.size(); ++h) {
        HouseAiRec& rec = ai_houses_[h];
        if (!rec.is_ai || !rec.cfg.team.enabled) continue;  // FIREWALL (spec §5)
        const AiTeamConfig& tc = rec.cfg.team;
        AiTeamState& st = rec.team_state;

        // 1) Member-death sweep [RE: ai-team.md §2]: prune dead/dying members
        //    (parallel vectors stay parallel); the pure lifecycle then sees
        //    the reopened slot (full=yes teams drop back to Recruiting).
        for (TeamInstance& t : st.teams) {
            for (int k = (int)t.members.size() - 1; k >= 0; --k) {
                const int m = t.members[k];
                if (m < 0 || m >= (int)units_.size() || !units_[m].alive ||
                    units_[m].die_time >= 0.0) {
                    t.members.erase(t.members.begin() + k);
                    t.member_types.erase(t.member_types.begin() + k);
                }
            }
        }

        // 2) Factory-lost cancel for the unit queues (the queues themselves
        //    already TICKED inside step_ai_houses's rec.queues.tick — no new
        //    production code) [RE: ai-team.md §6].
        if (rec.queues.producing(QueueCat::Infantry) != -1 &&
            factory_category_lost(tc.inf_factory_types, h))
            rec.queues.cancel(QueueCat::Infantry);
        if (rec.queues.producing(QueueCat::Vehicle) != -1 &&
            factory_category_lost(tc.veh_factory_types, h))
            rec.queues.cancel(QueueCat::Vehicle);
        rec.credits = rec.queues.credits();

        // 3) Launch READY unit slots (Infantry then Vehicle — deterministic).
        ai_launch_unit_ready(h, rec, QueueCat::Infantry);
        ai_launch_unit_ready(h, rec, QueueCat::Vehicle);

        // 4) Two-clock autocreate [RE-GATE APPLIED: ai-team.md §4]: TeamDelays
        //    (frames, per-difficulty) gates the management CHECK pass; within a
        //    check pass, AutocreateTime (minutes) gates the actual SPAWN; the
        //    TotalAITeamCap hard cap is enforced inside select_autocreate_team.
        //    At most ONE new team per spawn, highest priority first, deterministic.
        st.check_clock += dt;
        const double check_period = (double)tc.rules.team_delay_frames / 15.0;
        if (st.check_clock >= check_period) {
            st.check_clock = 0.0;
            st.spawn_clock += check_period;
            const double spawn_period = tc.rules.autocreate_time_min * 60.0;  // min->s
            if (st.spawn_clock >= spawn_period) {
                st.spawn_clock = 0.0;
                const int pick = select_autocreate_team(tc.aimd, tc.house_name,
                                                        st.teams, rec.credits, tc.rules);
                if (pick >= 0) {
                    TeamInstance t;
                    t.teamtype = pick;
                    st.teams.push_back(t);
                    ++st.teams_created;
                }
            }
        }

        // 5) Recruit-idle first [RE: ai-team.md §2] (re-scanned per team so a
        //    unit claimed by an earlier team is not offered to a later one),
        //    then ONE production intent per house tick through the EXISTING
        //    queues — tab2 (Infantry) / tab3 (Vehicle) routing [RE: §6].
        for (TeamInstance& t : st.teams) {
            if (t.state != TeamState::Recruiting) continue;
            recruit_idle(t, tc.aimd, ai_idle_units(h, rec));
        }
        const int want = next_produce_type(st.teams, tc.aimd, rec.credits,
                                           rec.cfg.mods.cost_mul, tc.unit_roster,
                                           tc.rules);
        if (want >= 0) {
            const bool veh = tc.unit_roster[want].is_vehicle;
            const QueueCat cat = veh ? QueueCat::Vehicle : QueueCat::Infantry;
            const std::vector<int>& ft = veh ? tc.veh_factory_types : tc.inf_factory_types;
            const bool factory = !factory_category_lost(ft, h);
            // BuildSlowdown OVER-QUEUE damp [RE-GATE APPLIED: ai-team.md §6]: on
            // Normal/Difficult, do NOT start another of a category the house
            // already has on-hand. The wanted type's queue is already guarded
            // (producing == -1); the damp additionally blocks if a unit of the
            // wanted type is already standing on-hand (launched, not yet
            // team-consumed). BuildSlowdown is a difficulty BYTE, NOT a build-
            // time multiplier — there is no speed penalty on start().
            bool onhand_of_type = false;
            for (const Unit& u : units_)
                if (u.alive && u.die_time < 0.0 && u.house == h &&
                    ai_unit_type_of(rec, u) == want) { onhand_of_type = true; break; }
            if (rec.queues.producing(cat) == -1 && factory &&
                !build_slowdown_blocks(tc.rules, onhand_of_type) &&
                can_start_unit(rec.queues.state(cat), want, tc.unit_roster,
                               buildings_, rec.cfg.roster, h, rec.cfg.groups,
                               factory)) {
                // Scalar start + slot restamp — the AI-1 building idiom: the
                // difficulty cost handicap on the charge and the BuildTime
                // handicap on the speed. No BuildSlowdown time-mul.
                const int cost = tc.unit_roster[want].cost;
                const int charge = std::max(1, (int)(cost * rec.cfg.mods.cost_mul));
                rec.queues.set_credits(rec.credits);
                start(rec.queues.state(cat), cost, charge,
                      rec.cfg.build_speed * rec.cfg.mods.build_time_mul);
                rec.queues.state(cat).slot.type = want;
                rec.credits = rec.queues.credits();
            }
        }

        // 6) Step every team (pure brain) + execute its command (impure
        //    glue); Disbanding teams release survivors (they simply stop
        //    being members -> idle, re-recruitable) and are erased
        //    [RE: ai-team.md §2].
        for (int i = 0; i < (int)st.teams.size();) {
            TeamInstance& t = st.teams[i];
            TeamView view;
            ai_fill_team_view(h, rec, t, view);
            const TeamCommand c =
                step_team(t, view, tc.aimd, st.rng, dt, tc.rules.multiplay);
            ai_execute_team_command(h, rec, t, c);
            if (t.state == TeamState::Disbanding)
                st.teams.erase(st.teams.begin() + i);
            else
                ++i;
        }
    }
}
```

```cpp
// FNV hash of the AI-2 end state for --demo-ai-team's two-run byte-identity
// gate [RE: ai-team.md]: the SORTED live-team rows {teamtype, state, pc,
// member count} + each team's SORTED member {cell, hp} rows + the
// produced-unit list (house units: image chars, cell, hp, alive) + credits +
// teams_created. A NEW function; world_hash/econ_hash/prod_hash/ai_base_hash
// UNTOUCHED.
uint64_t ai_team_hash(const ra2engine::World& world, int house) {
    uint64_t h = 1469598103934665603ULL;  // FNV offset basis
    if (const ra2engine::AiTeamState* st = world.ai_team_state(house)) {
        h = fnv_mix(h, static_cast<uint64_t>(st->teams_created));
        std::vector<std::array<int, 4>> rows;
        for (const ra2engine::TeamInstance& t : st->teams)
            rows.push_back({t.teamtype, (int)t.state, t.script_pc, (int)t.members.size()});
        std::sort(rows.begin(), rows.end());
        for (const auto& r : rows)
            for (int x : r) h = fnv_mix(h, static_cast<uint64_t>(x));
        for (const ra2engine::TeamInstance& t : st->teams) {
            std::vector<std::array<int, 3>> mem;
            for (int m : t.members) {
                if (m < 0 || m >= (int)world.units().size()) continue;
                const ra2engine::Unit& u = world.units()[m];
                mem.push_back({u.cell_x, u.cell_y, u.hp});
            }
            std::sort(mem.begin(), mem.end());
            for (const auto& r : mem)
                for (int x : r) h = fnv_mix(h, static_cast<uint64_t>(x));
        }
    }
    for (const ra2engine::Unit& u : world.units()) {   // produced-unit list
        if (u.house != house) continue;
        for (char ch : u.image) h = fnv_mix(h, static_cast<uint64_t>((unsigned char)ch));
        h = fnv_mix(h, static_cast<uint64_t>(u.cell_x));
        h = fnv_mix(h, static_cast<uint64_t>(u.cell_y));
        h = fnv_mix(h, static_cast<uint64_t>(u.hp));
        h = fnv_mix(h, static_cast<uint64_t>(u.alive ? 1 : 0));
    }
    h = fnv_mix(h, static_cast<uint64_t>(std::llround(world.ai_credits(house))));
    return h;
}
```
