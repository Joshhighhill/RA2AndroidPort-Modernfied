> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Non-player houses get a base-planning brain (sub-phase AI-1 of the three-sub-phase AI effort). The enemy Soviet house — which today owns a pre-built base and a working economy but never acts — gains a planner that decides what to build next (recovered urgency ordering), where to place it (recovered adjacency scan), and rebuilds what it loses, driving the EXISTING production queues, placement validity, and building-placed path. Verified by a new 6th permanent byte-identity gate `--demo-ai-base`.

**Architecture:** Task 1 recovers ground truth from gamemd.exe + rulesmd `[AI]`/difficulty sections into `docs/re/ai-base.md`; then the change lands additively, protecting all five byte-identity gates. A new pure module `engine/ai_base.{hpp,cpp}` holds the brain as free functions (`next_build`, `find_build_location`, `difficulty_mods`) over a read-only `BaseView` plus a small per-house `AiBaseState`; `engine/world.{hpp,cpp}` gains a per-house AI registry and `step_ai_houses()` (early-return firewall when no house is AI-flagged) that drives the EXISTING `ProductionQueues::start` / `placement_check` / `add_building` machinery — NO new production or placement code; `main.cpp` flags house 1 AI in the interactive path, expands the Soviet roster art, and wires the `--demo-ai-base` gate. // consumes a read-only BaseView the caller (World) fills, plus a per-house
// AiBaseState. doctest-testable. EVERY numeric constant below is a HUNT
// OUTPUT recorded in docs/re/ai-base.md; none is a guess — tests assert

namespace ra2engine {

// --- Recovered [AI] constants [RE-GATE APPLIED: ai-base.md §2 — pre-hunt
// values from the TS-lineage rulesmd keys; Task 1 confirms/corrects each] ---
constexpr double kAiPowerSurplus = 50.0;            // [AI] PowerSurplus= [RE: ai-base.md §2]
constexpr double kAiPowerEmergencyFraction = 0.75;  // [AI] PowerEmergencyFraction= [RE: ai-base.md §2]
constexpr int kAiBaseSizeAdd = 20;                  // [AI] BaseSizeAdd= [RE: ai-base.md §2]
constexpr double kAiBuildDelaySecs = 4.0;           // BuildDelay build-attempt cadence, seconds [RE: ai-base.md §1]
constexpr int kAiRefineryTarget = 1;                // refinery count target [RE: ai-base.md §2]
constexpr int kAiHarvestersPerRefinery = 1;         // harvester/refinery ratio [RE: ai-base.md §2]
// Per-difficulty defense-count targets, indexed by AiDifficulty
// [RE-GATE APPLIED: ai-base.md §2/§4 — pre-hunt placeholders].
constexpr int kAiDefenseTarget[3] = {2, 4, 6};
// Placement scan cap (Chebyshev radius from the base center)
// [RE-FALLBACK: ai-base.md §3 — engine bound, not a binary constant].
constexpr int kAiMaxScanRadius = 40;

// --- Recovered difficulty-group multipliers, indexed by AiDifficulty
// [RE-GATE APPLIED: ai-base.md §4 — 1.0 PLACEHOLDERS pre-hunt; the hunt MUST
// fill these from the three rulesmd difficulty groups (Cost=/BuildTime=/
// BuildDelay=) and pin the group-name -> AiDifficulty mapping] ---
constexpr double kDifficultyCost[3] = {1.0, 1.0, 1.0};
constexpr double kDifficultyBuildTime[3] = {1.0, 1.0, 1.0};
constexpr double kDifficultyBuildDelay[3] = {1.0, 1.0, 1.0};

enum class AiDifficulty { Easy = 0, Medium = 1, Hard = 2 };

// Raw multipliers of ONE rules difficulty group, filled by main.cpp from
// rulesmd via Rules::section_double [RE: ai-base.md §4]. Tests hand-build
// them from the kDifficulty* constants above.
struct DifficultyGroupRaw {
    double cost = 1.0;         // Cost=  (build cost multiplier)
    double build_time = 1.0;   // BuildTime= (build time multiplier)
    double build_delay = 1.0;  // BuildDelay= (AI build-attempt pacing multiplier)
};

// Resolved per-house difficulty handicaps [RE: ai-base.md §4].
struct DifficultyMods {
    double cost_mul = 1.0;   // multiplies the credits charged for a build
    double time_mul = 1.0;   // multiplies build TIME (applied to BuildSpeed)
    double delay_mul = 1.0;  // multiplies the build-clock gate
    int defense_target = 0;  // defenses the AI keeps standing
    int base_size_cap = 0;   // planned-base node cap (BaseSizeAdd-derived)
};

// The [AI]-section knobs, defaulting to the recovered constants (main.cpp
// re-reads them from the real rulesmd; tests use the defaults).
struct AiRules {
    double power_surplus = kAiPowerSurplus;
    double power_emergency_fraction = kAiPowerEmergencyFraction;
    int base_size_add = kAiBaseSizeAdd;
    double build_delay_secs = kAiBuildDelaySecs;
    int refinery_target = kAiRefineryTarget;
    int harvesters_per_refinery = kAiHarvestersPerRefinery;
};

// mods for difficulty d: multipliers from groups[d], defense_target from
// kAiDefenseTarget[d], base_size_cap = ai.base_size_add [RE-GATE APPLIED:
// ai-base.md §2/§4 — pre-hunt the cap is BaseSizeAdd flat; the original
// computes enemy-base-size + BaseSizeAdd, amend if the hunt pins that].
DifficultyMods difficulty_mods(AiDifficulty d, const DifficultyGroupRaw (&groups)[3],
                               const AiRules& ai);

// One planned base node [RE: ai-base.md §3/§5]: intended type + cell, the
// World building index while standing, and the destroyed-awaiting-rebuild
// flag (set by World when the placed building dies).
struct PlannedNode {
    int type = -1;        // roster index
    int cx = 0, cy = 0;   // intended foundation origin cell
    int building = -1;    // World building index once placed (-1 = not placed)
    bool dead = false;    // placed building destroyed -> rebuild target
};

// Per-house planner state. `pending[q]` is the plan index the building
// currently in queue q (0=Structure, 1=Defense) will fill on placement.
// `rng` is the engine-local xorshift32 the economy/production phases
// adjudicated [RE-FALLBACK: ai-base.md deviations — used ONLY where the hunt
// shows the original consulting the lockstep RNG; unused otherwise].
struct AiBaseState {
    AiDifficulty difficulty = AiDifficulty::Medium;
    double build_clock = 1.0e9;  // secs since last structure start; starts
                                 // "elapsed" so the first attempt fires
                                 // immediately [RE-GATE APPLIED: ai-base.md §1]
    std::vector<PlannedNode> plan;
    std::uint32_t rng = 0x2545F491u;  // deterministic fixed seed
    int pending[2] = {-1, -1};
};

// xorshift32 step (same algorithm family the economy phase adjudicated).
std::uint32_t ai_rand(std::uint32_t& s);

// Read-only snapshot of one house's base, filled by the caller (World).
struct BaseBuildingView {
    int type = -1;
    int cx = 0, cy = 0, fw = 1, fh = 1;
    bool operational = false;  // alive && Idle (fully built)
};
struct BaseView {
    std::vector<BaseBuildingView> own;  // this house's ALIVE buildings
    int center_x = 0, center_y = 0;     // base center = ConYard origin cell
    int threat_x = 0, threat_y = 0;     // defense bias target (enemy centroid)
    double credits = 0.0;
    int power_output = 0;
    int power_drain = 0;
    int harvesters = 0;                 // own alive harvesters
};

// Roster indices of the AI's build set, resolved once by main.cpp
// [RE-GATE APPLIED: ai-base.md §2 — the recovered [AI] build lists name the
// real ids]. -1 / empty = unavailable in this install (art missing): the
// planner SKIPS that need (art-guard-drop, never a crash).
struct AiCatalog {
    int power = -1;        // NAPOWR
    int refinery = -1;     // NAREFN
    int barracks = -1;     // NAHAND
    int war_factory = -1;  // NAWEAP
    std::vector<int> defenses;  // recovered defense set, preference order
};

// next_build's answer. type >= 0: start this roster type now (affordable,
// catalog-available). wait: a need was chosen but is unaffordable at
// cost*cost_mul -> bank income, start NOTHING cheaper [RE: ai-base.md §2].
// rebuild_node >= 0: this start re-fills plan[rebuild_node] at its cell.
struct BuildDecision {
    int type = -1;
    int rebuild_node = -1;
    bool wait = false;
};

// The recovered urgency ordering [RE-GATE APPLIED: ai-base.md §2], evaluated
// top-down, first satisfied need wins: rebuild -> power (emergency OR
// surplus top-up) -> economy (refinery/harvester ratio) -> production
// prereqs (barracks then war factory) -> defenses (< per-difficulty target)
// -> expansion (< base_size_cap). The build clock is gated by the CALLER
// (World owns dt); affordability (credits >= cost * cost_mul) is gated here.
BuildDecision next_build(const AiBaseState& state, const BaseView& view,
                         const AiCatalog& cat,
                         const std::vector<BuildingTypeDef>& roster,
                         const AiRules& ai, const DifficultyMods& mods);

// Deterministic placement scan [RE-GATE APPLIED: ai-base.md §3; RE-FALLBACK:
// nearest-first Chebyshev rings from the base center, row-major within a
// ring; defenses probe each ring's cells nearest the threat first].
// `can_place(cx, cy)` is the EXISTING player placement validity bound by the
// caller (placement_check: foundation fully buildable + within Adjacent of
// an owned structure + no overlap). Returns the first accepted foundation
// origin, or {-1, -1} when no cell within kAiMaxScanRadius passes (boxed
// in -> the caller holds and retries).
Cellxy find_build_location(const BaseView& view, bool is_defense,
                           const std::function<bool(int, int)>& can_place);

}  // namespace ra2engine

namespace ra2engine {

std::uint32_t ai_rand(std::uint32_t& s) {
    // xorshift32 [RE-FALLBACK: ai-base.md deviations — engine-local RNG for
    // any adjudicated tie-break; deterministic, never wall-clock].
    s ^= s << 13;

    m.base_size_cap = ai.base_size_add;      // [RE-GATE APPLIED: ai-base.md §2]
    return m;
}

namespace {
long long dist2(int ax, int ay, int bx, int by) {
    const long long dx = ax - bx, dy = ay - by;
    return dx * dx + dy * dy;
}
}  // namespace

Cellxy find_build_location(const BaseView& view, bool is_defense,
                           const std::function<bool(int, int)>& can_place) {
    // [RE-GATE APPLIED: ai-base.md §3 — recovered Find_Build_Location scan.
    // RE-FALLBACK adjudicated pre-hunt: nearest-first Chebyshev rings from
    // the base center, row-major (dy then dx ascending) within a ring —
    // deterministic, foundation/adjacency/overlap all enforced by can_place
    // (the player's placement_check, reused verbatim). Defenses probe each
    // ring's cells nearest the THREAT first (perimeter-facing bias,
    // BaseBias/AISafeDistance stand-in) with the row-major order as the
    // stable tie-break.]
    const int ox = view.center_x, oy = view.center_y;
    for (int r = 0; r <= kAiMaxScanRadius; ++r) {
        std::vector<Cellxy> ring;
        ring.reserve(r == 0 ? 1 : 8 * r);
        for (int dy = -r; dy <= r; ++dy)
            for (int dx = -r; dx <= r; ++dx) {
                if (std::max(std::abs(dx), std::abs(dy)) != r) continue;
                ring.push_back(Cellxy{ox + dx, oy + dy});
            }
        if (is_defense) {
            std::stable_sort(ring.begin(), ring.end(), [&](const Cellxy& a, const Cellxy& b) {
                return dist2(a.x, a.y, view.threat_x, view.threat_y) <
                       dist2(b.x, b.y, view.threat_x, view.threat_y);
            });
        }
        for (const Cellxy& c : ring)
            if (can_place(c.x, c.y)) return c;
    }
    return Cellxy{-1, -1};  // boxed in / nothing within the cap -> caller holds
}

    //    harvester/refinery ratio; a refinery ships its free harvester].

        return pick(cat.barracks, view, roster, mods);
    if (cat.war_factory >= 0 && count_of(view, cat.war_factory) == 0)
        return pick(cat.war_factory, view, roster, mods);

    const int defenses = count_defenses(view, cat);

    //    §2 — the real expansion draws from the recovered [AI] build lists /
    //    BaseNode plan; RE-FALLBACK pre-hunt: alternate power plant and
    //    defense, deterministically by plan size].

BaseView World::make_ai_base_view(int house, const HouseAiRec& rec) const {
    BaseView v;
    bool conyard_center = false;  // a ConYard's cell locked the center
    bool any_center = false;      // some building set a provisional center
    long long tx = 0, ty = 0, tn = 0;
    for (const Building& b : buildings_) {
        if (!b.alive || b.state == Building::State::Dead) continue;
        if (b.house == house) {
            BaseBuildingView bb;
            bb.type = b.type;
            bb.cx = b.cx; bb.cy = b.cy; bb.fw = b.fw; bb.fh = b.fh;
            bb.operational = b.state == Building::State::Idle;
            v.own.push_back(bb);
            // Base center = the FIRST ConYard's origin cell; if the house
            // owns none (yet), the FIRST building stands in.
            const bool is_cy = b.type >= 0 && b.type < (int)rec.cfg.roster.size() &&
                               rec.cfg.roster[b.type].conyard;
            if (is_cy && !conyard_center) {
                v.center_x = b.cx; v.center_y = b.cy;
                conyard_center = true;
            } else if (!conyard_center && !any_center) {
                v.center_x = b.cx; v.center_y = b.cy;
                any_center = true;
            }
        } else {
            tx += b.cx; ty += b.cy; ++tn;  // threat = other houses' centroid
        }
    }
    if (tn > 0) { v.threat_x = (int)(tx / tn); v.threat_y = (int)(ty / tn); }
    else { v.threat_x = grid_w_ / 2; v.threat_y = grid_h_ / 2; }
    v.credits = rec.credits;
    const PowerTotals pt = power_totals(buildings_, rec.cfg.roster, house);
    v.power_output = pt.output;
    v.power_drain = pt.drain;
    for (const Unit& u : units_)
        if (u.alive && u.house == house && u.is_harvester) ++v.harvesters;
    return v;
}

// Places queue `cat`'s READY building at its planned node's cell through the
// EXISTING validity + add_building path [RE: ai-base.md §3/§6]. Returns true
// if the slot was consumed (placed); false = HOLD (retry next frame).
bool World::ai_place_ready(int house, HouseAiRec& rec, QueueCat cat) {
    const int q = cat == QueueCat::Structure ? 0 : 1;
    if (!rec.queues.ready(cat)) return false;
    const int node_i = rec.state.pending[q];
    if (node_i < 0 || node_i >= (int)rec.state.plan.size()) {
        rec.queues.placed(cat);  // defensive: orphaned ready slot
        rec.state.pending[q] = -1;
        return true;
    }
    PlannedNode& node = rec.state.plan[node_i];
    const BuildingTypeDef& t = rec.cfg.roster[node.type];
    auto can_place = [&](int cx, int cy) {
        return placement_check(cx, cy, t.fw, t.fh, house, t.adjacent, passable_,
                               grid_w_, grid_h_, occ_, buildings_,
                               /*bypass_adjacency=*/false).ok;
    };
    int px = node.cx, py = node.cy;
    if (!can_place(px, py)) {
        // Something moved onto the node's cell: re-scan (deterministic).
        const BaseView v = make_ai_base_view(house, rec);
        const bool is_def = std::find(rec.cfg.catalog.defenses.begin(),
                                      rec.cfg.catalog.defenses.end(),
                                      node.type) != rec.cfg.catalog.defenses.end();
        const Cellxy alt = find_build_location(v, is_def, can_place);
        if (alt.x < 0) return false;  // boxed in -> HOLD the ready slot
        px = alt.x; py = alt.y;
        node.cx = px; node.cy = py;
    }
    const double buildup = node.type < (int)rec.cfg.buildup_totals.size()
                               ? rec.cfg.buildup_totals[node.type] : 0.0;
    const int idx = add_building(node.type, px, py, t.fw, t.fh, house, t.strength,
                                 t.armor_ix, buildup, /*instant=*/false);
    node.building = idx;
    node.dead = false;
    rec.queues.placed(cat);
    rec.state.pending[q] = -1;
    return true;
}

        if (!rec.is_ai) continue;
        const AiHouseConfig& cfg = rec.cfg;

        // 1) Income (the economy phase accumulates per-house unload income).
        rec.credits += take_income(h);

        // 2) Node-death detection [RE: ai-base.md §5].
        for (PlannedNode& n : rec.state.plan) {
            if (n.building < 0) continue;
            const Building& b = buildings_[n.building];
            if (!b.alive || b.state == Building::State::Dead) {
                n.building = -1;
                n.dead = true;
            }
        }

        // 3) ConYard-lost cancels both building queues (the same rule the
        //    player's prod_tick applies) [RE: ai-base.md §6].
        bool conyard = false;
        for (const Building& b : buildings_) {
            if (b.alive && b.state == Building::State::Idle && b.house == h &&
                b.type >= 0 && b.type < (int)cfg.roster.size() &&
                cfg.roster[b.type].conyard) { conyard = true; break; }
        }
        if (!conyard) {
            for (QueueCat c : {QueueCat::Structure, QueueCat::Defense})
                if (rec.queues.producing(c) != -1) rec.queues.cancel(c);
        }

        // 4) Tick the queues: SAME power clamp as the player [RE: ai-base.md
        //    §6], difficulty time handicap applied at start() (below).
        const PowerTotals pt = power_totals(buildings_, cfg.roster, h);
        const double pmul = build_speed_multiplier(pt, cfg.low_power_penalty);
        rec.queues.set_credits(rec.credits);
        const double sp[4] = {1.0, 1.0, 1.0, 1.0};  // one ConYard in AI-1 — no
                                                    // MultipleFactory speedup
        rec.queues.tick(dt, pmul, sp, cfg.roster);
        rec.credits = rec.queues.credits();

        // 5) Place READY slots (Structure then Defense — deterministic order).
        ai_place_ready(h, rec, QueueCat::Structure);
        ai_place_ready(h, rec, QueueCat::Defense);

        // 6) Build clock [RE-GATE APPLIED: ai-base.md §1 — BuildDelay x the
        //    difficulty delay_mul between structure starts].
        rec.state.build_clock += dt;
        if (rec.state.build_clock < cfg.ai.build_delay_secs * cfg.mods.delay_mul)
            continue;
        if (!conyard) continue;  // no factory: nothing can start

        // 7) Decide.
        const BaseView view = make_ai_base_view(h, rec);
        const BuildDecision d =
            next_build(rec.state, view, cfg.catalog, cfg.roster, cfg.ai, cfg.mods);
        if (d.type < 0) continue;  // nothing needed, or wait/bank

        // 8) Route to the queue the player would use [RE: ai-base.md §6]:
        //    defenses -> Defense queue, everything else -> Structure queue.
        const bool is_def = std::find(cfg.catalog.defenses.begin(),
                                      cfg.catalog.defenses.end(),
                                      d.type) != cfg.catalog.defenses.end();
        const QueueCat cat = is_def ? QueueCat::Defense : QueueCat::Structure;
        const int q = is_def ? 1 : 0;
        if (rec.queues.producing(cat) != -1 || rec.state.pending[q] != -1)
            continue;  // slot busy (single-slot, as today)

        // 9) Same prereq/tech/factory gate as the player (can_start —
        //    EXISTING function) [RE: ai-base.md §6].
        if (!can_start(rec.queues.state(cat), d.type, cfg.roster, buildings_, h,
                       cfg.groups, conyard))
            continue;

        // 10) Choose the location NOW and record the node [RE: ai-base.md §3]
        //     (rebuild reuses its stored cell; ai_place_ready re-validates).
        int node_i;
        if (d.rebuild_node >= 0) {
            node_i = d.rebuild_node;
        } else {
            const BuildingTypeDef& t = cfg.roster[d.type];
            auto can_place = [&](int cx, int cy) {
                return placement_check(cx, cy, t.fw, t.fh, h, t.adjacent, passable_,
                                       grid_w_, grid_h_, occ_, buildings_,
                                       /*bypass_adjacency=*/false).ok;
            };
            const Cellxy loc = find_build_location(view, is_def, can_place);
            if (loc.x < 0) continue;  // boxed in: try again next clock
            PlannedNode n;
            n.type = d.type;
            n.cx = loc.x; n.cy = loc.y;
            rec.state.plan.push_back(n);
            node_i = (int)rec.state.plan.size() - 1;
        }

        // 11) START on the EXISTING queue economy: difficulty cost handicap
        //     on the charge, time handicap on the speed [RE: ai-base.md §4].
        const BuildingTypeDef& t = cfg.roster[d.type];
        const int basis = t.charge > 0 ? t.charge : t.cost;
        const int ai_charge =
            std::max(1, (int)(basis * cfg.mods.cost_mul));
        rec.queues.set_credits(rec.credits);
        start(rec.queues.state(cat), t.cost, ai_charge,
              cfg.build_speed * cfg.mods.time_mul);
        rec.queues.state(cat).slot.type = d.type;
        rec.credits = rec.queues.credits();
        rec.state.pending[q] = node_i;
        rec.state.build_clock = 0.0;
    }
}

uint64_t ai_base_hash(const ra2engine::World& world, int house,
                      const std::vector<ra2engine::BuildingTypeDef>& roster) {
    uint64_t h = 1469598103934665603ULL;  // FNV offset basis
    std::vector<std::array<int, 4>> rows;
    for (const ra2engine::Building& b : world.buildings()) {
        if (!b.alive || b.house != house) continue;
        rows.push_back({b.type, b.cx, b.cy, b.hp});
