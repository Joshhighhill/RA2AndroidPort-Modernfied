cmake_minimum_required(VERSION 3.20)
project(ra2_render_stability C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SDL2_SRC_DIR)
    message(FATAL_ERROR "Set SDL2_SRC_DIR to an existing SDL 2.30.9 source directory")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDL2_SRC_DIR}" sdl)
add_executable(render_stability main.cpp)
target_include_directories(render_stability PRIVATE ../../engine)
target_link_libraries(render_stability PRIVATE SDL2-static)
